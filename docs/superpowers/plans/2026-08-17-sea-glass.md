# Sea Glass Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Sea Glass: she fills one painted silhouette with frosted color chips. Easy 9 tinted holes, Medium 16 painted holes plus extras, Hard 25 copy-the-mosaic plus extras.

**Architecture:** `glass.js` owns pictures, layouts, extras, `place`, and `playAgain` (Node-testable). `app.js` owns tray, drag/tap, wiggle, and cheer. Pictures and chips are painted PNGs in the existing Beach Day crayon style — never SVG clip-art.

**Tech Stack:** HTML, CSS, vanilla JS. Tests: `node test/glass.test.js` (same `test` / `assert` / `always` harness as `test/puzzle.test.js`). No new libraries.

## Global Constraints

- No timer, no losing score, no required reading.
- Color-only matching. No peel after a correct place. No pinch-zoom. No undo.
- Easy 9 / Medium 16 / Hard 25. One of 8 pictures at random.
- Easy cue `tint` and 0 extras. Medium cue `paint` and 2 extras. Hard cue `copy` and 3 extras.
- Cheer: `What a picture!` Play again keeps difficulty, new picture roll.
- Art is painted PNGs matching the kid, shells, and puzzle scenes. No generic SVG, icon fonts, outline-only clip-art, or stock photos.
- Tidy, Puzzle, and Path rules do not change.
- Default difficulty medium; unknown → medium.
- `random` is `[0, 1)`; omitted → `Math.random`.
- Menu order: Tide Pool Tidy, Beach Puzzle, Crab Path, Sea Glass.

```
PICTURE_COUNT: 8
PICTURES: ["heart","sun","fish","starfish","boat","flower","crab","moon"]
COLORS: ["white","seafoam","aqua","cobalt","amber"]
grid(difficulty) -> { holes: 9|16|25 }
extraCount(difficulty) -> 0|2|3
cueFor(difficulty) -> "tint"|"paint"|"copy"
createSession({ difficulty, random }) -> session
pictureId(session) -> string
pictureColors(pictureIndex) -> string[]
layout(pictureIndex, difficulty) -> [{ id, color, x, y }]
place(session, pieceId, holeId) -> { ok, session }
pieceById(session, id) -> piece | null
playAgain(session, random) -> session
```

Session shape:

```
{
  difficulty,     // "easy" | "medium" | "hard"
  pictureIndex,   // 0..7
  cue,            // "tint" | "paint" | "copy"
  holes,          // [{ id, color }]
  filled,         // { [holeId]: piece | null }
  tray,           // [{ id, color }]
  complete
}
```

Piece `{ id, color }` with integer `id` starting at `0`. `place` / `playAgain` copy; they do not mutate the input.

```js
/**
 * Sea Glass — pictures, layouts, extras, place.
 * Works in Node (tests) and the browser (window.SeaGlass).
 */
(function (root) {
  "use strict";

  var PICTURES = ["heart", "sun", "fish", "starfish", "boat", "flower", "crab", "moon"];
  var PICTURE_COUNT = PICTURES.length;
  var COLORS = ["white", "seafoam", "aqua", "cobalt", "amber"];

  var PICTURE_COLORS = [
    ["aqua", "white", "seafoam"],
    ["amber", "white"],
    ["aqua", "seafoam", "white"],
    ["amber", "white"],
    ["cobalt", "white", "amber"],
    ["seafoam", "amber", "white"],
    ["amber", "white", "seafoam"],
    ["white", "cobalt"]
  ];

  // [easy, medium, hard] each { color: count }
  var COUNTS = [
    [
      { aqua: 5, white: 2, seafoam: 2 },
      { aqua: 8, white: 4, seafoam: 4 },
      { aqua: 13, white: 6, seafoam: 6 }
    ],
    [
      { amber: 6, white: 3 },
      { amber: 11, white: 5 },
      { amber: 17, white: 8 }
    ],
    [
      { aqua: 5, seafoam: 2, white: 2 },
      { aqua: 8, seafoam: 4, white: 4 },
      { aqua: 13, seafoam: 6, white: 6 }
    ],
    [
      { amber: 6, white: 3 },
      { amber: 11, white: 5 },
      { amber: 17, white: 8 }
    ],
    [
      { cobalt: 4, white: 3, amber: 2 },
      { cobalt: 7, white: 5, amber: 4 },
      { cobalt: 11, white: 8, amber: 6 }
    ],
    [
      { seafoam: 5, amber: 2, white: 2 },
      { seafoam: 8, amber: 4, white: 4 },
      { seafoam: 13, amber: 6, white: 6 }
    ],
    [
      { amber: 5, white: 2, seafoam: 2 },
      { amber: 8, white: 4, seafoam: 4 },
      { amber: 13, white: 6, seafoam: 6 }
    ],
    [
      { white: 6, cobalt: 3 },
      { white: 11, cobalt: 5 },
      { white: 17, cobalt: 8 }
    ]
  ];

  function normalizeDifficulty(value) {
    if (value === "easy" || value === "hard" || value === "medium") return value;
    return "medium";
  }

  function normalizePicture(index) {
    var i = Number(index);
    if (i !== i || i < 0 || i >= PICTURE_COUNT) return 0;
    return i;
  }

  function grid(difficulty) {
    var d = normalizeDifficulty(difficulty);
    if (d === "easy") return { holes: 9 };
    if (d === "hard") return { holes: 25 };
    return { holes: 16 };
  }

  function extraCount(difficulty) {
    var d = normalizeDifficulty(difficulty);
    if (d === "easy") return 0;
    if (d === "hard") return 3;
    return 2;
  }

  function cueFor(difficulty) {
    var d = normalizeDifficulty(difficulty);
    if (d === "easy") return "tint";
    if (d === "hard") return "copy";
    return "paint";
  }

  function countIndex(difficulty) {
    var d = normalizeDifficulty(difficulty);
    if (d === "easy") return 0;
    if (d === "hard") return 2;
    return 1;
  }

  function gridPositions(n) {
    var side = n === 9 ? 3 : n === 25 ? 5 : 4;
    var inset = 0.16;
    var cells = [];
    var r;
    var c;
    for (r = 0; r < side; r++) {
      for (c = 0; c < side; c++) {
        cells.push({
          x: inset + ((c + 0.5) * (1 - 2 * inset)) / side,
          y: inset + ((r + 0.5) * (1 - 2 * inset)) / side
        });
      }
    }
    return cells;
  }

  function colorSequence(pictureIndex, difficulty) {
    var counts = COUNTS[normalizePicture(pictureIndex)][countIndex(difficulty)];
    var order = PICTURE_COLORS[normalizePicture(pictureIndex)];
    var seq = [];
    var i;
    var n;
    for (i = 0; i < order.length; i++) {
      for (n = 0; n < counts[order[i]]; n++) seq.push(order[i]);
    }
    return seq;
  }

  function layout(pictureIndex, difficulty) {
    var seq = colorSequence(pictureIndex, difficulty);
    var spots = gridPositions(seq.length);
    return seq.map(function (color, i) {
      return { id: i, color: color, x: spots[i].x, y: spots[i].y };
    });
  }

  function clonePiece(p) {
    return { id: p.id, color: p.color };
  }

  function cloneSession(s) {
    var filled = {};
    var id;
    for (id in s.filled) {
      if (Object.prototype.hasOwnProperty.call(s.filled, id)) {
        filled[id] = s.filled[id] ? clonePiece(s.filled[id]) : null;
      }
    }
    return {
      difficulty: s.difficulty,
      pictureIndex: s.pictureIndex,
      cue: s.cue,
      holes: s.holes.map(function (h) {
        return { id: h.id, color: h.color };
      }),
      filled: filled,
      tray: s.tray.map(clonePiece),
      complete: s.complete
    };
  }

  function pickIndex(n, random) {
    if (n <= 0) return 0;
    var i = Math.floor(random() * n);
    if (i < 0) return 0;
    if (i >= n) return n - 1;
    return i;
  }

  function shuffle(list, random) {
    var arr = list.slice();
    var i;
    var j;
    var t;
    for (i = arr.length - 1; i > 0; i--) {
      j = pickIndex(i + 1, random);
      t = arr[i];
      arr[i] = arr[j];
      arr[j] = t;
    }
    return arr;
  }

  function createSession(opts) {
    opts = opts || {};
    var difficulty = normalizeDifficulty(opts.difficulty);
    var random = typeof opts.random === "function" ? opts.random : Math.random;
    var pictureIndex = pickIndex(PICTURE_COUNT, random);
    var laid = layout(pictureIndex, difficulty);
    var holes = laid.map(function (h) {
      return { id: h.id, color: h.color };
    });
    var filled = {};
    var tray = [];
    var nextId = 0;
    var i;
    for (i = 0; i < holes.length; i++) {
      filled[holes[i].id] = null;
      tray.push({ id: nextId, color: holes[i].color });
      nextId += 1;
    }
    var used = {};
    PICTURE_COLORS[pictureIndex].forEach(function (c) {
      used[c] = true;
    });
    var unused = COLORS.filter(function (c) {
      return !used[c];
    });
    unused = shuffle(unused, random);
    var extras = extraCount(difficulty);
    var extraPieces = [];
    for (i = 0; i < extras; i++) {
      var color;
      if (i < unused.length) color = unused[i];
      else color = extraPieces[0].color;
      extraPieces.push({ id: nextId, color: color });
      nextId += 1;
    }
    tray = shuffle(tray.concat(extraPieces), random);
    return {
      difficulty: difficulty,
      pictureIndex: pictureIndex,
      cue: cueFor(difficulty),
      holes: holes,
      filled: filled,
      tray: tray,
      complete: false
    };
  }

  function pictureId(session) {
    return PICTURES[normalizePicture(session.pictureIndex)];
  }

  function pictureColors(pictureIndex) {
    return PICTURE_COLORS[normalizePicture(pictureIndex)].slice();
  }

  function pieceById(session, id) {
    var i;
    for (i = 0; i < session.tray.length; i++) {
      if (session.tray[i].id === id) return clonePiece(session.tray[i]);
    }
    var key;
    for (key in session.filled) {
      if (session.filled[key] && session.filled[key].id === id) {
        return clonePiece(session.filled[key]);
      }
    }
    return null;
  }

  function holeById(session, holeId) {
    var i;
    for (i = 0; i < session.holes.length; i++) {
      if (session.holes[i].id === holeId) return session.holes[i];
    }
    return null;
  }

  function place(session, pieceId, holeId) {
    var next = cloneSession(session);
    var hole = holeById(next, holeId);
    if (!hole) return { ok: false, session: next };
    if (next.filled[hole.id]) return { ok: false, session: next };
    var idx = -1;
    var i;
    for (i = 0; i < next.tray.length; i++) {
      if (next.tray[i].id === pieceId) idx = i;
    }
    if (idx === -1) return { ok: false, session: next };
    var piece = next.tray[idx];
    if (piece.color !== hole.color) return { ok: false, session: next };
    next.tray = next.tray.slice(0, idx).concat(next.tray.slice(idx + 1));
    next.filled[hole.id] = clonePiece(piece);
    var empty = 0;
    for (i = 0; i < next.holes.length; i++) {
      if (!next.filled[next.holes[i].id]) empty += 1;
    }
    next.complete = empty === 0;
    return { ok: true, session: next };
  }

  function playAgain(session, random) {
    if (!session) return createSession({ random: random });
    return createSession({ difficulty: session.difficulty, random: random });
  }

  var SeaGlass = {
    PICTURE_COUNT: PICTURE_COUNT,
    PICTURES: PICTURES.slice(),
    COLORS: COLORS.slice(),
    grid: grid,
    extraCount: extraCount,
    cueFor: cueFor,
    createSession: createSession,
    pictureId: pictureId,
    pictureColors: pictureColors,
    layout: layout,
    place: place,
    pieceById: pieceById,
    playAgain: playAgain
  };

  if (typeof module !== "undefined" && module.exports) {
    module.exports = SeaGlass;
  } else {
    root.SeaGlass = SeaGlass;
  }
})(typeof window !== "undefined" ? window : typeof global !== "undefined" ? global : this);
```

```js
function startGlass(difficulty) {
  activeGame = "glass";
  glassSelected = null;
  hide(menuScreen);
  hide(difficultyScreen);
  hide(playScreen);
  hide(pathScreen);
  hide(puzzleScreen);
  show(glassScreen);
  hideCheer();
  startGlassDeal(SeaGlass.createSession({ difficulty: difficulty }));
}

function startGlassDeal(next) {
  glassSession = next;
  glassSelected = null;
  renderGlass();
}

function renderGlass() {
  var id = SeaGlass.pictureId(glassSession);
  var laid = SeaGlass.layout(glassSession.pictureIndex, glassSession.difficulty);
  var kind = glassSession.cue === "paint" ? "paint" : "outline";
  glassBoard.style.backgroundImage = "url('" + SeaGlass.pictureSrc(id, kind) + "')";
  glassBoard.innerHTML = "";
  laid.forEach(function (spot) {
    var hole = document.createElement("button");
    hole.type = "button";
    hole.className = "glass-hole";
    hole.dataset.hole = String(spot.id);
    hole.style.left = spot.x * 100 + "%";
    hole.style.top = spot.y * 100 + "%";
    var filled = glassSession.filled[spot.id];
    if (filled) {
      hole.classList.add("is-filled");
      hole.innerHTML =
        '<img src="' + SeaGlass.chipSrc(filled.color) + '" alt="" draggable="false" />';
    } else if (glassSession.cue === "tint") {
      hole.classList.add("is-tint");
      hole.style.backgroundColor = glassTint(spot.color);
    }
    glassBoard.appendChild(hole);
  });

  if (glassSession.cue === "copy") {
    glassCopy.hidden = false;
    glassCopy.innerHTML =
      '<img src="' + SeaGlass.pictureSrc(id, "done") + '" alt="" />';
  } else {
    glassCopy.hidden = true;
    glassCopy.innerHTML = "";
  }

  glassTray.innerHTML = "";
  glassSession.tray.forEach(function (piece) {
    var btn = document.createElement("button");
    btn.type = "button";
    btn.className = "glass-chip";
    btn.dataset.piece = String(piece.id);
    if (glassSelected === piece.id) btn.classList.add("selected");
    btn.innerHTML =
      '<img src="' + SeaGlass.chipSrc(piece.color) + '" alt="" draggable="false" />';
    glassTray.appendChild(btn);
    bindGlassDrag(btn, piece);
  });
}

function glassTint(color) {
  return {
    white: "#f4f1ea",
    seafoam: "#8fd4b8",
    aqua: "#3ec6c4",
    cobalt: "#2a5cad",
    amber: "#e8a03a"
  }[color];
}
```

```js
function tryGlassDrop(pieceId, holeEl) {
  if (!holeEl) return false;
  var holeId = Number(holeEl.dataset.hole);
  var result = SeaGlass.place(glassSession, pieceId, holeId);
  glassSession = result.session;
  glassSelected = result.ok ? null : pieceId;

  if (result.ok && glassSession.complete) afterGlass();
  return true;
}
```
